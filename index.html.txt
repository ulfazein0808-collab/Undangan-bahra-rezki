// Countdown
const eventDate=new Date("Feb 07, 2026 10:00:00").getTime();
setInterval(()=>{
const now=new Date().getTime();
const d=eventDate-now;
if(d<0){countdown.innerHTML="Acara sedang berlangsung 🎉";}
else{
const day=Math.floor(d/(1000*60*60*24));
const h=Math.floor((d%(1000*60*60*24))/(1000*60*60));
const m=Math.floor((d%(1000*60*60))/(1000*60));
const s=Math.floor((d%(1000*60))/1000);
countdown.innerHTML=`Menuju acara: ${day} hari ${h} jam ${m} menit ${s} detik`;
}},1000);
